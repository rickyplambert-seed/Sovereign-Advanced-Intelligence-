import os
import json
import time
import subprocess

LOG_PATH = os.path.expanduser('~/signal.log')
THREAT_LOG = os.path.expanduser('~/threats.log')
UI_STATUS_PATH = os.path.expanduser('~/aegis_ui_status.json')
COMPILER_PATH = os.path.expanduser('~/compile_resources.py')
TEMP_THRESHOLD = 45.0

def inspect_latest_signal():
    if not os.path.exists(LOG_PATH):
        return

    try:
        with open(LOG_PATH, 'rb') as f:
            f.seek(-2, os.SEEK_END)
            while f.read(1) != b'\n':
                f.seek(-2, os.SEEK_CUR)
            last_line = f.readline().decode('utf-8').strip()
    except OSError:
        with open(LOG_PATH, 'r') as f:
            last_line = f.readlines()[-1].strip()

    if not last_line:
        return

    ui_theme_state = "SAFE_BASELINE"
    status_color = "#00E676"  # Operational Green
    active_alert = None

    try:
        record = json.loads(last_line)
        node_data = record.get("data", {})
        node_name = node_data.get("node", "Unknown")
        metrics = node_data.get("metrics", {})
        cpu_temp = metrics.get("cpu_temp", 0)

        if cpu_temp > TEMP_THRESHOLD:
            ui_theme_state = "CRITICAL_THERMAL"
            status_color = "#FF1744"  # Critical Crimson
            active_alert = {
                "alert_timestamp": int(time.time()),
                "type": "THERMAL_ANOMALY",
                "node": node_name,
                "details": f"Temperature breach: {cpu_temp}°C"
            }
        elif not record.get("integrity_verified", True):
            ui_theme_state = "UNVERIFIED_STREAM_WARN"
            status_color = "#FFEA00"  # Warning Amber
            active_alert = {
                "alert_timestamp": int(time.time()),
                "type": "UNVERIFIED_STREAM",
                "node": node_name,
                "details": "Payload bypassed structured JSON compilation rules."
            }
    except:
        return

    if active_alert:
        try:
            with open(THREAT_LOG, 'a') as f:
                f.write(json.dumps(active_alert) + '\n')
        except:
            pass

    ui_matrix = {
        "ui_sync_timestamp": int(time.time()),
        "system_status_mode": ui_theme_state,
        "theme_vectors": {
            "status_bar_color": status_color,
            "accent_glow": status_color,
            "widget_border_alpha": "0.85" if active_alert else "0.40"
        }
    }

    try:
        with open(UI_STATUS_PATH, 'w') as f:
            json.dump(ui_matrix, f, indent=4)

        # Trigger Resource Compiler if script exists
        if os.path.exists(COMPILER_PATH):
            subprocess.Popen(['python3', COMPILER_PATH], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    except:
        pass

if __name__ == '__main__':
    inspect_latest_signal()
